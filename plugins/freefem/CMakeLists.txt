# The SDK-dependent adapter is a separate optional module, never part of core.
enable_language(CXX)
if(FORTSPARSE_ENABLE_SUPERLU OR FORTSPARSE_ENABLE_MUMPS)
    message(FATAL_ERROR "The native-only FreeFem module requires FORTSPARSE_ENABLE_SUPERLU=OFF and FORTSPARSE_ENABLE_MUMPS=OFF")
endif()
if(NOT FORTSPARSE_NATIVE_AMD_FOUND)
    message(FATAL_ERROR "The native FreeFem module requires optional AMD ordering (FORTSPARSE_ENABLE_AMD=ON with AMD installed)")
endif()
find_program(FREEFEM_CXX_EXECUTABLE NAMES ff-c++ REQUIRED)
find_program(FREEFEM_EXECUTABLE NAMES FreeFem++ REQUIRED)
# Dry-run only: consume SDK include/ABI definitions, never vendor linker flags.
execute_process(COMMAND "${FREEFEM_CXX_EXECUTABLE}" -n -c
    "${CMAKE_CURRENT_SOURCE_DIR}/fortsparse_freefem.cpp"
    OUTPUT_VARIABLE _ff_compile ERROR_VARIABLE _ff_error RESULT_VARIABLE _ff_status)
if(NOT _ff_status EQUAL 0)
    message(FATAL_ERROR "Cannot read FreeFem SDK compile contract: ${_ff_error}")
endif()
string(REGEX MATCH "[^\n]+" _ff_compile "${_ff_compile}")
separate_arguments(_ff_arguments UNIX_COMMAND "${_ff_compile}")
set(_ff_includes)
set(_ff_definitions)
foreach(_argument IN LISTS _ff_arguments)
    if(_argument MATCHES "^-I(.+)")
        list(APPEND _ff_includes "${CMAKE_MATCH_1}")
    elseif(_argument MATCHES "^-D(.+)")
        list(APPEND _ff_definitions "${CMAKE_MATCH_1}")
    endif()
endforeach()
# Derive the include directory anew when switching ff-c++/runtime prefixes.
unset(_ff_sdk_include_dir)
find_path(_ff_sdk_include_dir ff++.hpp HINTS ${_ff_includes} REQUIRED NO_CACHE NO_DEFAULT_PATH)
set(FREEFEM_INCLUDE_DIR "${_ff_sdk_include_dir}")
# Exclude umbrella-header instantiations of unused vendor adapters, preserving
# the installed SDK's ABI definitions and leaving the installation untouched.
# Debian/Ubuntu SDKs export FreeFEM-config.h; other installations export config.h.
# Wrap the actual advertised headers, without inventing configuration macros.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/sdk-config")
set(_ff_config_found FALSE)
foreach(_ff_config_name config.h FreeFEM-config.h)
    file(REMOVE "${CMAKE_CURRENT_BINARY_DIR}/sdk-config/${_ff_config_name}")
    unset(_ff_config_file)
    find_file(_ff_config_file "${_ff_config_name}" HINTS "${FREEFEM_INCLUDE_DIR}" ${_ff_includes}
        NO_CACHE NO_DEFAULT_PATH)
    if(_ff_config_file)
        set(_ff_config_found TRUE)
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sdk-config/${_ff_config_name}"
            "#pragma once\n#include \"${_ff_config_file}\"\n#undef HAVE_LIBUMFPACK\n#undef HAVE_LIBCHOLMOD\n")
    endif()
endforeach()
if(NOT _ff_config_found)
    message(FATAL_ERROR "FreeFem SDK exports neither config.h nor FreeFEM-config.h in its include paths")
endif()
add_library(fortsparse_freefem MODULE fortsparse_freefem.cpp)
target_include_directories(fortsparse_freefem BEFORE PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/sdk-config")
target_include_directories(fortsparse_freefem SYSTEM PRIVATE ${_ff_includes} "${FREEFEM_INCLUDE_DIR}")
target_compile_definitions(fortsparse_freefem PRIVATE ${_ff_definitions})
target_compile_features(fortsparse_freefem PRIVATE cxx_std_14)
target_link_libraries(fortsparse_freefem PRIVATE fortsparse)
set_target_properties(fortsparse_freefem PROPERTIES PREFIX "" OUTPUT_NAME fortsparse_native
    SUFFIX ".so" LIBRARY_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/lib/freefem")
if(APPLE)
    target_link_options(fortsparse_freefem PRIVATE "LINKER:-undefined,dynamic_lookup")
endif()
install(TARGETS fortsparse_freefem LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}/fortsparse/freefem")
if(FORTSPARSE_BUILD_TESTING)
    add_subdirectory(test)
endif()
